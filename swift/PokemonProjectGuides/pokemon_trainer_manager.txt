// Single-file demo for UIKit "Pokémon Trainer Manager"
// - Models (Trainer, Pokemon, Badge)
// - JSON bundle loading
// - UITableViewController with sections by Region
// - Trainer detail with list of Pokémon and delegation to Pokemon detail
// - Add Trainer form (basic)
// - QLPreviewController integration for previewing badges / trainer card
// - Simple leaderboard & filter & battle timer

import UIKit
import QuickLook

// MARK: - Models

enum Region: String, Codable, CaseIterable {
    case kanto = "Kanto"
    case johto = "Johto"
    case hoenn = "Hoenn"
    case sinnoh = "Sinnoh"
    case unova = "Unova"
    case kalos = "Kalos"
    case alola = "Alola"
    case galar = "Galar"
}

struct Badge: Codable {
    let name: String
    let imageName: String // image stored in bundle / assets
}

struct Pokemon: Codable {
    let id: Int
    let name: String
    let types: [String]
    let imageName: String
    let level: Int
}

struct Trainer: Codable {
    let id: Int
    var name: String
    var photoName: String?
    var region: Region
    var badges: [Badge]
    var pokemons: [Pokemon]
    var score: Int // for leaderboard
}

// MARK: - Data Provider (Bundle JSON)

class DataProvider {
    static let shared = DataProvider()
    private(set) var trainers: [Trainer] = []

    private init() {
        loadFromBundle()
    }

    func loadFromBundle() {
        guard let url = Bundle.main.url(forResource: "trainers", withExtension: "json") else {
            print("trainers.json not found in bundle")
            trainers = []
            return
        }

        do {
            let data = try Data(contentsOf: url)
            let decoder = JSONDecoder()
            decoder.keyDecodingStrategy = .convertFromSnakeCase
            trainers = try decoder.decode([Trainer].self, from: data)
        } catch {
            print("Failed to decode trainers.json: \(error)")
            trainers = []
        }
    }

    func addTrainer(_ trainer: Trainer) {
        trainers.append(trainer)
    }
}

// MARK: - Protocols

protocol PokemonSelectionDelegate: AnyObject {
    func didSelectPokemon(_ pokemon: Pokemon, from trainer: Trainer)
}

// MARK: - Main Table (Trainers by Region)

class TrainersTableViewController: UITableViewController {
    private var trainersByRegion: [Region: [Trainer]] = [:]
    private var regions: [Region] = []

    private var filterPokemonType: String? = nil

    override func viewDidLoad() {
        super.viewDidLoad()
        title = "Dresseurs"
        navigationItem.rightBarButtonItem = UIBarButtonItem(barButtonSystemItem: .add, target: self, action: #selector(addTrainerTapped))
        navigationItem.leftBarButtonItem = UIBarButtonItem(title: "Filtres", style: .plain, target: self, action: #selector(filterTapped))
        tableView.register(UITableViewCell.self, forCellReuseIdentifier: "cell")
        reloadData()
    }

    func reloadData() {
        let trainers = DataProvider.shared.trainers
        trainersByRegion = [:]
        for r in Region.allCases { trainersByRegion[r] = [] }
        for t in trainers {
            if let filter = filterPokemonType {
                if t.pokemons.contains(where: { $0.types.contains { $0.lowercased() == filter.lowercased() } }) {
                    trainersByRegion[t.region, default: []].append(t)
                }
            } else {
                trainersByRegion[t.region, default: []].append(t)
            }
        }
        regions = Region.allCases.filter { (trainersByRegion[$0]?.isEmpty == false) }
        tableView.reloadData()
    }

    // MARK: Table Data Source

    override func numberOfSections(in tableView: UITableView) -> Int {
        return regions.count
    }

    override func tableView(_ tableView: UITableView, titleForHeaderInSection section: Int) -> String? {
        return regions[section].rawValue
    }

    override func tableView(_ tableView: UITableView, numberOfRowsInSection section: Int) -> Int {
        let region = regions[section]
        return trainersByRegion[region]?.count ?? 0
    }

    override func tableView(_ tableView: UITableView, cellForRowAt indexPath: IndexPath) -> UITableViewCell {
        let cell = tableView.dequeueReusableCell(withIdentifier: "cell", for: indexPath)
        let region = regions[indexPath.section]
        let trainer = trainersByRegion[region]![indexPath.row]
        cell.textLabel?.text = trainer.name + " (\(trainer.pokemons.count) Pokémon)"
        cell.accessoryType = .disclosureIndicator
        if let photo = trainer.photoName {
            cell.imageView?.image = UIImage(named: photo)
        } else {
            cell.imageView?.image = UIImage(systemName: "person.circle")
        }
        return cell
    }

    // MARK: Table Delegate

    override func tableView(_ tableView: UITableView, didSelectRowAt indexPath: IndexPath) {
        let region = regions[indexPath.section]
        let trainer = trainersByRegion[region]![indexPath.row]
        let detail = TrainerDetailViewController(trainer: trainer)
        detail.pokemonDelegate = self
        navigationController?.pushViewController(detail, animated: true)
    }

    // MARK: Actions

    @objc func addTrainerTapped() {
        let add = AddTrainerViewController()
        add.delegate = self
        let nav = UINavigationController(rootViewController: add)
        present(nav, animated: true)
    }

    @objc func filterTapped() {
        let alert = UIAlertController(title: "Filtrer par type", message: "Entrez un type (ex: Fire, Water)", preferredStyle: .alert)
        alert.addTextField { tf in tf.placeholder = "Type" }
        alert.addAction(UIAlertAction(title: "OK", style: .default, handler: { _ in
            let text = alert.textFields?.first?.text?.trimmingCharacters(in: .whitespacesAndNewlines)
            self.filterPokemonType = (text?.isEmpty ?? true) ? nil : text
            self.reloadData()
        }))
        alert.addAction(UIAlertAction(title: "Reset", style: .destructive, handler: { _ in
            self.filterPokemonType = nil
            self.reloadData()
        }))
        alert.addAction(UIAlertAction(title: "Cancel", style: .cancel))
        present(alert, animated: true)
    }
}

extension TrainersTableViewController: AddTrainerDelegate {
    func didAddTrainer(_ trainer: Trainer) {
        DataProvider.shared.addTrainer(trainer)
        reloadData()
    }
}

extension TrainersTableViewController: PokemonSelectionDelegate {
    func didSelectPokemon(_ pokemon: Pokemon, from trainer: Trainer) {
        let detail = PokemonDetailViewController(pokemon: pokemon)
        navigationController?.pushViewController(detail, animated: true)
    }
}

// MARK: - Trainer Detail

class TrainerDetailViewController: UIViewController {
    var trainer: Trainer
    weak var pokemonDelegate: PokemonSelectionDelegate?

    private let imageView = UIImageView()
    private let nameLabel = UILabel()
    private let badgesCollection = UICollectionView(frame: .zero, collectionViewLayout: UICollectionViewFlowLayout())
    private let tableView = UITableView()

    init(trainer: Trainer) {
        self.trainer = trainer
        super.init(nibName: nil, bundle: nil)
    }

    required init?(coder: NSCoder) { fatalError("init(coder:) has not been implemented") }

    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .systemBackground
        title = trainer.name

        setupHeader()
        setupTable()
        navigationItem.rightBarButtonItem = UIBarButtonItem(title: "Prévisualiser", style: .plain, target: self, action: #selector(previewTrainerCard))
    }

    func setupHeader() {
        imageView.translatesAutoresizingMaskIntoConstraints = false
        imageView.contentMode = .scaleAspectFill
        imageView.layer.cornerRadius = 40
        imageView.layer.masksToBounds = true
        if let photo = trainer.photoName { imageView.image = UIImage(named: photo) }

        nameLabel.translatesAutoresizingMaskIntoConstraints = false
        nameLabel.text = "\(trainer.name) — \(trainer.pokemons.count) Pokémon"
        nameLabel.font = UIFont.preferredFont(forTextStyle: .headline)

        view.addSubview(imageView)
        view.addSubview(nameLabel)

        NSLayoutConstraint.activate([
            imageView.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor, constant: 16),
            imageView.leadingAnchor.constraint(equalTo: view.leadingAnchor, constant: 16),
            imageView.widthAnchor.constraint(equalToConstant: 80),
            imageView.heightAnchor.constraint(equalToConstant: 80),

            nameLabel.centerYAnchor.constraint(equalTo: imageView.centerYAnchor),
            nameLabel.leadingAnchor.constraint(equalTo: imageView.trailingAnchor, constant: 12),
            nameLabel.trailingAnchor.constraint(equalTo: view.trailingAnchor, constant: -16),
        ])
    }

    func setupTable() {
        tableView.translatesAutoresizingMaskIntoConstraints = false
        tableView.dataSource = self
        tableView.delegate = self
        tableView.register(UITableViewCell.self, forCellReuseIdentifier: "pokemonCell")
        view.addSubview(tableView)

        NSLayoutConstraint.activate([
            tableView.topAnchor.constraint(equalTo: imageView.bottomAnchor, constant: 16),
            tableView.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            tableView.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            tableView.bottomAnchor.constraint(equalTo: view.bottomAnchor),
        ])
    }

    @objc func previewTrainerCard() {
        // Create a local file URL for a generated trainer card image or use an asset
        // For demo, assume there's an image named "trainer_card_\(trainer.id)"
        let imageName = "trainer_card_\(trainer.id)"
        if let url = Bundle.main.url(forResource: imageName, withExtension: "png") {
            let ql = QLPreviewController()
            ql.dataSource = QuickLookDataSource(urls: [url])
            present(ql, animated: true)
        } else {
            let alert = UIAlertController(title: "Aucune carte", message: "Aucune carte HD disponible pour ce dresseur dans le bundle.", preferredStyle: .alert)
            alert.addAction(UIAlertAction(title: "OK", style: .default))
            present(alert, animated: true)
        }
    }
}

extension TrainerDetailViewController: UITableViewDataSource, UITableViewDelegate {
    func tableView(_ tableView: UITableView, numberOfRowsInSection section: Int) -> Int { trainer.pokemons.count }

    func tableView(_ tableView: UITableView, cellForRowAt indexPath: IndexPath) -> UITableViewCell {
        let cell = tableView.dequeueReusableCell(withIdentifier: "pokemonCell", for: indexPath)
        let p = trainer.pokemons[indexPath.row]
        cell.textLabel?.text = "\(p.name) — Lv. \(p.level)"
        cell.imageView?.image = UIImage(named: p.imageName) ?? UIImage(systemName: "bolt.circle")
        cell.accessoryType = .detailButton
        return cell
    }

    func tableView(_ tableView: UITableView, didSelectRowAt indexPath: IndexPath) {
        let p = trainer.pokemons[indexPath.row]
        pokemonDelegate?.didSelectPokemon(p, from: trainer)
    }

    // detail button -> preview badge or pokemon image
    func tableView(_ tableView: UITableView, accessoryButtonTappedForRowWith indexPath: IndexPath) {
        let p = trainer.pokemons[indexPath.row]
        if let url = Bundle.main.url(forResource: p.imageName, withExtension: "png") {
            let ql = QLPreviewController()
            ql.dataSource = QuickLookDataSource(urls: [url])
            present(ql, animated: true)
        }
    }
}

// MARK: - QuickLook helper

class QuickLookDataSource: NSObject, QLPreviewControllerDataSource {
    let urls: [URL]
    init(urls: [URL]) { self.urls = urls }
    func numberOfPreviewItems(in controller: QLPreviewController) -> Int { urls.count }
    func previewController(_ controller: QLPreviewController, previewItemAt index: Int) -> QLPreviewItem { urls[index] as QLPreviewItem }
}

// MARK: - Pokemon Detail

class PokemonDetailViewController: UIViewController {
    let pokemon: Pokemon
    private let imageView = UIImageView()
    private let nameLabel = UILabel()
    private let typesLabel = UILabel()
    private var battleTimer: Timer?
    private var remainingSeconds = 0

    init(pokemon: Pokemon) {
        self.pokemon = pokemon
        super.init(nibName: nil, bundle: nil)
    }
    required init?(coder: NSCoder) { fatalError() }

    override func viewDidLoad() {
        super.viewDidLoad()
        title = pokemon.name
        view.backgroundColor = .systemBackground

        imageView.translatesAutoresizingMaskIntoConstraints = false
        imageView.image = UIImage(named: pokemon.imageName)
        imageView.contentMode = .scaleAspectFit

        nameLabel.translatesAutoresizingMaskIntoConstraints = false
        nameLabel.text = "Lv. \(pokemon.level) — \(pokemon.name)"
        nameLabel.font = UIFont.preferredFont(forTextStyle: .headline)

        typesLabel.translatesAutoresizingMaskIntoConstraints = false
        typesLabel.text = "Types: \(pokemon.types.joined(separator: ", "))"

        let battleButton = UIButton(type: .system)
        battleButton.translatesAutoresizingMaskIntoConstraints = false
        battleButton.setTitle("Simuler combat 10s", for: .normal)
        battleButton.addTarget(self, action: #selector(startBattleTimer), for: .touchUpInside)

        view.addSubview(imageView)
        view.addSubview(nameLabel)
        view.addSubview(typesLabel)
        view.addSubview(battleButton)

        NSLayoutConstraint.activate([
            imageView.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor, constant: 20),
            imageView.centerXAnchor.constraint(equalTo: view.centerXAnchor),
            imageView.widthAnchor.constraint(equalToConstant: 180),
            imageView.heightAnchor.constraint(equalToConstant: 180),

            nameLabel.topAnchor.constraint(equalTo: imageView.bottomAnchor, constant: 12),
            nameLabel.centerXAnchor.constraint(equalTo: view.centerXAnchor),

            typesLabel.topAnchor.constraint(equalTo: nameLabel.bottomAnchor, constant: 8),
            typesLabel.centerXAnchor.constraint(equalTo: view.centerXAnchor),

            battleButton.topAnchor.constraint(equalTo: typesLabel.bottomAnchor, constant: 16),
            battleButton.centerXAnchor.constraint(equalTo: view.centerXAnchor),
        ])
    }

    @objc func startBattleTimer() {
        remainingSeconds = 10
        battleTimer?.invalidate()
        battleTimer = Timer.scheduledTimer(timeInterval: 1, target: self, selector: #selector(timerTick), userInfo: nil, repeats: true)
        let alert = UIAlertController(title: "Combat démarré", message: "10s restants", preferredStyle: .alert)
        alert.addAction(UIAlertAction(title: "Stop", style: .destructive, handler: { _ in self.stopBattle() }))
        present(alert, animated: true)
    }

    @objc func timerTick() {
        remainingSeconds -= 1
        if remainingSeconds <= 0 {
            stopBattle()
            dismiss(animated: true) {
                let done = UIAlertController(title: "Combat terminé", message: "Le combat est terminé !", preferredStyle: .alert)
                done.addAction(UIAlertAction(title: "OK", style: .default))
                UIApplication.shared.windows.first?.rootViewController?.present(done, animated: true)
            }
        } else {
            // update presented alert message if needed
        }
    }

    func stopBattle() {
        battleTimer?.invalidate()
        battleTimer = nil
        remainingSeconds = 0
    }
}

// MARK: - Add Trainer Form

protocol AddTrainerDelegate: AnyObject {
    func didAddTrainer(_ trainer: Trainer)
}

class AddTrainerViewController: UIViewController, UINavigationControllerDelegate {
    weak var delegate: AddTrainerDelegate?

    private let nameField = UITextField()
    private var selectedImageName: String? = nil
    private let regionPicker = UIPickerView()
    private var region: Region = .kanto

    override func viewDidLoad() {
        super.viewDidLoad()
        title = "Ajouter un dresseur"
        view.backgroundColor = .systemGroupedBackground
        navigationItem.leftBarButtonItem = UIBarButtonItem(barButtonSystemItem: .cancel, target: self, action: #selector(cancel))
        navigationItem.rightBarButtonItem = UIBarButtonItem(barButtonSystemItem: .save, target: self, action: #selector(save))

        setupForm()
    }

    func setupForm() {
        nameField.translatesAutoresizingMaskIntoConstraints = false
        nameField.placeholder = "Nom du dresseur"
        nameField.borderStyle = .roundedRect

        let chooseImage = UIButton(type: .system)
        chooseImage.translatesAutoresizingMaskIntoConstraints = false
        chooseImage.setTitle("Choisir photo (bundle)", for: .normal)
        chooseImage.addTarget(self, action: #selector(pickImageFromBundle), for: .touchUpInside)

        regionPicker.translatesAutoresizingMaskIntoConstraints = false
        regionPicker.dataSource = self
        regionPicker.delegate = self

        view.addSubview(nameField)
        view.addSubview(chooseImage)
        view.addSubview(regionPicker)

        NSLayoutConstraint.activate([
            nameField.topAnchor.constraint(equalTo: view.safeAreaLayoutGuide.topAnchor, constant: 20),
            nameField.leadingAnchor.constraint(equalTo: view.leadingAnchor, constant: 16),
            nameField.trailingAnchor.constraint(equalTo: view.trailingAnchor, constant: -16),

            chooseImage.topAnchor.constraint(equalTo: nameField.bottomAnchor, constant: 12),
            chooseImage.centerXAnchor.constraint(equalTo: view.centerXAnchor),

            regionPicker.topAnchor.constraint(equalTo: chooseImage.bottomAnchor, constant: 12),
            regionPicker.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            regionPicker.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            regionPicker.heightAnchor.constraint(equalToConstant: 120),
        ])
    }

    @objc func pickImageFromBundle() {
        // For demo: show a small list of images from assets
        let alert = UIAlertController(title: "Choisir photo", message: nil, preferredStyle: .actionSheet)
        let candidates = ["trainer_m.png", "trainer_f.png", "ash.png"]
        for c in candidates {
            alert.addAction(UIAlertAction(title: c, style: .default, handler: { _ in self.selectedImageName = c }))
        }
        alert.addAction(UIAlertAction(title: "Cancel", style: .cancel))
        present(alert, animated: true)
    }

    @objc func cancel() { dismiss(animated: true) }

    @objc func save() {
        guard let name = nameField.text, !name.isEmpty else {
            let a = UIAlertController(title: "Erreur", message: "Entrez un nom", preferredStyle: .alert)
            a.addAction(UIAlertAction(title: "OK", style: .default))
            present(a, animated: true)
            return
        }

        let new = Trainer(id: Int.random(in: 1000...9999), name: name, photoName: selectedImageName, region: region, badges: [], pokemons: [], score: 0)
        delegate?.didAddTrainer(new)
        dismiss(animated: true)
    }
}

extension AddTrainerViewController: UIPickerViewDataSource, UIPickerViewDelegate {
    func numberOfComponents(in pickerView: UIPickerView) -> Int { 1 }
    func pickerView(_ pickerView: UIPickerView, numberOfRowsInComponent component: Int) -> Int { Region.allCases.count }
    func pickerView(_ pickerView: UIPickerView, titleForRow row: Int, forComponent component: Int) -> String? { Region.allCases[row].rawValue }
    func pickerView(_ pickerView: UIPickerView, didSelectRow row: Int, inComponent component: Int) { region = Region.allCases[row] }
}
